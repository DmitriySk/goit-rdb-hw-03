DROP TABLE IF EXISTS hw3_taxi_trips CASCADE;

CREATE TABLE hw3_taxi_trips (
trip_id          BIGINT
GENERATED BY DEFAULT AS IDENTITY
PRIMARY KEY,

    vendor_id        SMALLINT,

    pickup_at        TIMESTAMPTZ NOT NULL,
    dropoff_at       TIMESTAMPTZ NOT NULL,

    passenger_count  SMALLINT
        CHECK (
            passenger_count IS NULL
            OR passenger_count >= 0
        ),

    trip_distance    NUMERIC(10, 2)
        CHECK (trip_distance >= 0),

    pu_location_id   INTEGER,
    do_location_id   INTEGER,

    payment_type     SMALLINT,

    fare_amount      NUMERIC(10, 2),
    tip_amount       NUMERIC(10, 2),
    total_amount     NUMERIC(10, 2),

    CHECK (dropoff_at >= pickup_at)
);



SQL: приклад очищення staging → final

WITH parsed AS (
SELECT
NULLIF(TRIM(vendor_id_raw), '')::SMALLINT AS vendor_id,

        NULLIF(TRIM(pickup_raw), '')::TIMESTAMP
            AT TIME ZONE 'America/New_York' AS pickup_at,

        NULLIF(TRIM(dropoff_raw), '')::TIMESTAMP
            AT TIME ZONE 'America/New_York' AS dropoff_at,

        NULLIF(TRIM(passenger_count_raw), '')::NUMERIC::SMALLINT
            AS passenger_count,

        NULLIF(TRIM(trip_distance_raw), '')::NUMERIC(10, 2)
            AS trip_distance,

        NULLIF(TRIM(pu_location_id_raw), '')::INTEGER
            AS pu_location_id,

        NULLIF(TRIM(do_location_id_raw), '')::INTEGER
            AS do_location_id,

        NULLIF(TRIM(payment_type_raw), '')::SMALLINT
            AS payment_type,

        NULLIF(TRIM(fare_amount_raw), '')::NUMERIC(10, 2)
            AS fare_amount,

        NULLIF(TRIM(tip_amount_raw), '')::NUMERIC(10, 2)
            AS tip_amount,

        NULLIF(TRIM(total_amount_raw), '')::NUMERIC(10, 2)
            AS total_amount
    FROM hw3_taxi_staging
)
INSERT INTO hw3_taxi_trips (
vendor_id,
pickup_at,
dropoff_at,
passenger_count,
trip_distance,
pu_location_id,
do_location_id,
payment_type,
fare_amount,
tip_amount,
total_amount
)
SELECT
vendor_id,
pickup_at,
dropoff_at,
passenger_count,
trip_distance,
pu_location_id,
do_location_id,
payment_type,
fare_amount,
tip_amount,
total_amount
FROM parsed
WHERE pickup_at IS NOT NULL
AND dropoff_at IS NOT NULL
AND dropoff_at >= pickup_at
AND trip_distance >= 0
AND (
passenger_count IS NULL
OR passenger_count >= 0
);

4.3. Один outlier-check

Достатньо виконати один із двох варіантів: статистичний або доменний outlier-check.

Якщо запит не повернув рядків, це допустимий результат — поясніть, що саме було перевірено.



SQL: правило 3σ для приблизно симетричного розподілу

WITH stats AS (
SELECT
AVG(numeric_col) AS mean_value,
STDDEV_SAMP(numeric_col) AS sd_value
FROM hw3_data_clean
WHERE numeric_col IS NOT NULL
)
SELECT
t.*,
ROUND(
((t.numeric_col - s.mean_value) / NULLIF(s.sd_value, 0))::NUMERIC,
2
) AS z_score
FROM hw3_data_clean AS t
CROSS JOIN stats AS s
WHERE t.numeric_col IS NOT NULL
AND s.sd_value > 0
AND ABS(t.numeric_col - s.mean_value) > 3 * s.sd_value
ORDER BY ABS(t.numeric_col - s.mean_value) DESC
LIMIT 20;

5. Десять DQL/EDA-запитів	Business-question / SQL / interpretation; покриття обов’язкових конструкцій Теми 3.	0–30
6. Reflection	Приблизно 200–250 слів про якість даних, features, leakage-ризики й покращення pipeline.	0–10

Критерії прийняття

Робота відповідає вимогам, якщо виконано всі наведені нижче умови.

У LMS додано посилання на публічний репозиторій goit-rdb-hw-03 та архів ДЗ3_ПІБ.zip.
Notebook виконується після Restart & Run All без ручного виправлення шляхів або SQL.
Перша секція запускає pgserver і показує версію PostgreSQL.
Дані отримуються з офіційного джерела відтворюваним способом.
CSV завантажено через COPY FROM STDIN у staging-таблицю.
Є окрема типізована таблиця та INSERT INTO ... SELECT із очищенням і приведенням типів.
DDL містить доречні типи, PRIMARY KEY, NOT NULL і щонайменше два предметні CHECK.
Показано кількість рядків у staging і clean, перевірку дублікатів та null/sentinel audit щонайменше п’яти колонок.
Виконано розподіл категорії, числову статистику й один outlier/domain-check.
Є щонайменше десять запитів у форматі business-question / SQL / interpretation.
У десяти запитах сукупно використано всі обов’язкові конструкції Теми 3.
Reflection відповідає на питання про якість, features, leakage і pipeline.
